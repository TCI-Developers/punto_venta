namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Venta;
use App\Models\CuentaPorPagar;

class InventarioController extends Controller
{
    public function index()
    {
        $inventario = [];

        // 1. Entradas desde cuentas por pagar
        $entradas = CuentaPorPagar::all()->flatMap(function ($cuenta) {
            $detalle = json_decode($cuenta->detalle, true); // campo con el JSON
            return collect([$detalle]);
        })->groupBy('code_product')->map(function ($items) {
            $first = $items->first();
            return [
                'codigo' => $first['code_product'],
                'descripcion' => $first['descripcion_producto'],
                'total_entrada' => collect($items)->sum('cant') ?? 1, // o el campo que represente la cantidad
                'costo_unitario' => $first['precio_unitario'],
            ];
        });

        // 2. Salidas desde ventas
        $salidas = Venta::all()->flatMap(function ($venta) {
            $detalle = json_decode($venta->detalle, true);
            return collect($detalle);
        })->groupBy('part_to_product_id')->map(function ($items) {
            return collect($items)->sum('cant') ?? 1;
        });

        // 3. Unir entradas y salidas
        foreach ($entradas as $codigo => $item) {
            $total_salida = $salidas[$codigo] ?? 0;
            $existencia_real = $item['total_entrada'] - $total_salida;

            $inventario[] = [
                'codigo' => $item['codigo'],
                'descripcion' => $item['descripcion'],
                'total_entrada' => $item['total_entrada'],
                'total_salida' => $total_salida,
                'existencia_real' => $existencia_real,
                'costo_unitario' => $item['costo_unitario'],
                'costo_inventario' => $existencia_real * $item['costo_unitario'],
            ];
        }

        return view('inventario.index', compact('inventario'));
    }
}
